function toast(msg, type) {
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.className = 'toast ' + (type || '') + ' show';
  setTimeout(() => t.className = 'toast', 3200);
}

function postAction(payload) {
  return fetch(GAS_URL, {
    method: 'POST', mode: 'no-cors',
    body: JSON.stringify(payload)
  }).catch(err => console.warn('POST err:', err));
}

function getAction(action) {
  return fetch(GAS_URL + '?action=' + encodeURIComponent(action))
    .then(r => r.json())
    .catch(err => { console.warn('GET err:', err); return null; });
}

function aqiCategory(aqi) {
  if (!aqi) return { label:'—', cls:'badge-purple' };
  if (aqi <= 50)  return { label:'Good',       cls:'badge-low' };
  if (aqi <= 100) return { label:'Moderate',   cls:'badge-low' };
  if (aqi <= 150) return { label:'Unhealthy',  cls:'badge-med' };
  if (aqi <= 200) return { label:'Unhealthy',  cls:'badge-high' };
  if (aqi <= 300) return { label:'Very Unhealthy', cls:'badge-high' };
  return { label:'Hazardous', cls:'badge-high' };
}

async function loadWeather() {
  const d = await getAction('weather');
  if (!d || d.error) return;

let sentinelOn = false, micStream = null, audioCtx = null, rafId = null;

document.getElementById('sentinel-btn').addEventListener('click', async () => {
  const txt = document.getElementById('sentinel-btn-text');
  const freqEl = document.getElementById('sentinel-freq');
  const barEl = document.getElementById('sentinel-bar');
  const statusEl = document.getElementById('sentinel-status-text');
  const micEl = document.getElementById('sentinel-mic-text');
  const badge = document.getElementById('sentinel-risk-badge');

  sentinelOn = !sentinelOn;

  if (sentinelOn) {
    txt.innerHTML = '<span class="spinner"></span> Arming…';
    try {
      micStream = await navigator.mediaDevices.getUserMedia({ audio: true });
      audioCtx = new (window.AudioContext || window.webkitAudioContext)();
      const src = audioCtx.createMediaStreamSource(micStream);
      const analyser = audioCtx.createAnalyser();
      analyser.fftSize = 128;
      src.connect(analyser);
      const data = new Uint8Array(analyser.frequencyBinCount);

      txt.textContent = '🔴 Disarm Sentinel';
      statusEl.textContent = 'Listening…';
      statusEl.style.color = 'var(--green)';
      micEl.textContent = 'Active';
      micEl.style.color = 'var(--green)';
      badge.textContent = 'Armed';
      badge.className = 'badge badge-high';

      const loop = () => {
        if (!sentinelOn) return;
        analyser.getByteFrequencyData(data);
        const maxVal = Math.max(...data);
        const peakIdx = data.indexOf(maxVal);
        const hz = Math.round(peakIdx / data.length * (audioCtx.sampleRate / 2));
        freqEl.textContent = hz + ' Hz';
        barEl.style.width = Math.min(100, maxVal / 2.55) + '%';

        if (maxVal > 200) {
          statusEl.textContent = '⚠ Anomaly!';
          statusEl.style.color = 'var(--red)';
          postAction({ action:'drone', drone_id:'MIC_01', alert_type:'AcousticAnomaly',
                       latitude:'22.5726', longitude:'88.3639', severity:'Medium' });
        } else {
          statusEl.textContent = 'Listening…';
          statusEl.style.color = 'var(--green)';
        }
        rafId = requestAnimationFrame(loop);
      };
      loop();
      toast('🎙️ Sentinel armed — mic active', 'success');
    } catch(e) {
      sentinelOn = false;
      txt.textContent = '📡 Arm Sentinel';
      statusEl.textContent = 'Mic denied';
      toast('Mic access denied — sentinel offline', 'error');
    }
  } else {
    txt.textContent = '📡 Arm Sentinel';
    freqEl.textContent = '-- Hz';
    barEl.style.width = '0%';
    statusEl.textContent = 'Inactive';
    statusEl.style.color = 'var(--muted)';
    micEl.textContent = 'Off';
    micEl.style.color = 'var(--muted)';
    badge.textContent = 'Standby';
    badge.className = 'badge badge-low';
    if (rafId) cancelAnimationFrame(rafId);
    if (micStream) micStream.getTracks().forEach(t => t.stop());
    if (audioCtx) audioCtx.close();
    micStream = audioCtx = null;
  }
});

document.addEventListener('mousemove', function(e) {
  const antenna = document.getElementById('antenna-group');
  const svg     = document.getElementById('satellite-3d');
  if (!antenna || !svg) return;

  const rect   = svg.getBoundingClientRect();
  // Antenna pivot is at SVG coordinate (60, 28) — map to screen space
  const pivotX = rect.left + (60 / 120) * rect.width;
  const pivotY = rect.top  + (28 / 100) * rect.height;

  const dx  = e.clientX - pivotX;
  const dy  = e.clientY - pivotY;
  // atan2 gives angle from pivot to cursor; subtract 90° so antenna points UP by default
  const deg = Math.atan2(dy, dx) * (180 / Math.PI) + 90;

  antenna.style.transform = `rotate(${deg}deg)`;
});

// Called by Search button or Enter key
async function searchByName(overrideQuery) {
  const query = overrideQuery ||
                document.getElementById('hero-location-search').value.trim() ||
                document.getElementById('location-search').value.trim();

  if (!query) { toast('⚠️ Enter a city or district name', 'error'); return; }

  const btn = document.querySelector('.hero-search-btn-go');
  const orig = btn ? btn.textContent : '';
  if (btn) btn.textContent = '⏳ Searching…';

  try {
    // Nominatim geocoding — name → lat/lon (free, no API key)
    const geoRes = await fetch(
      'https://nominatim.openstreetmap.org/search?q=' +
      encodeURIComponent(query + ', India') +
      '&format=json&limit=1&accept-language=en',
      { headers: { 'Accept-Language': 'en' } }
    );
    const geoData = await geoRes.json();

    if (!geoData || geoData.length === 0) {
      toast('❌ Location not found — try a different name', 'error');
      if (btn) btn.textContent = orig;
      return;
    }

    const lat = parseFloat(geoData[0].lat);
    const lon = parseFloat(geoData[0].lon);
    const displayName = geoData[0].display_name.split(',').slice(0, 2).join(',').trim();

    // Sync both search inputs
    document.getElementById('hero-location-search').value = displayName;
    const hdr = document.getElementById('location-search');
    if (hdr) hdr.value = displayName;

    toast('📍 Showing weather for ' + displayName, 'success');
    await searchLocation(lat, lon, displayName);
    if (typeof updateMapLocation === 'function') updateMapLocation(lat, lon, displayName);

  } catch (err) {
    toast('❌ Search failed — check connection', 'error');
    console.error('Search error:', err);
  }

  if (btn) btn.textContent = orig;
}

// Called by both searchByName and useMyLocation
// Fetches weather/AQI for given lat/lon and updates all stat cards
async function searchLocation(lat, lon, label) {
  try {
    // Fetch weather directly from Open-Meteo (same API your GAS uses)
    const [weatherRes, aqiRes] = await Promise.all([
      fetch(
        'https://api.open-meteo.com/v1/forecast?latitude=' + lat + '&longitude=' + lon +
        '&current=temperature_2m,apparent_temperature,relative_humidity_2m,wind_speed_10m,weather_code'
      ).then(r => r.json()),
      fetch(
        'https://air-quality-api.open-meteo.com/v1/air-quality?latitude=' + lat +
        '&longitude=' + lon + '&current=pm10,pm2_5,us_aqi'
      ).then(r => r.json())
    ]);

    const w = weatherRes.current;
    const q = aqiRes.current;

    // Update stat strip
    document.getElementById('live-temp').textContent     = Math.round(w.temperature_2m)    + '°C';
    document.getElementById('live-feels').textContent    = Math.round(w.apparent_temperature) + '°C';
    document.getElementById('live-humidity').textContent = Math.round(w.relative_humidity_2m) + '%';
    document.getElementById('live-wind').textContent     = Math.round(w.wind_speed_10m);
    document.getElementById('live-updated').textContent  = new Date().toLocaleTimeString('en-IN', { hour: '2-digit', minute: '2-digit' });

    if (q && q.us_aqi != null) {
      const aqi = Math.round(q.us_aqi);
      document.getElementById('live-aqi').textContent      = aqi;
      document.getElementById('aqi-bar-text').textContent  = aqi + ' / 500';
      document.getElementById('aqi-bar').style.width       = Math.min(100, aqi / 5) + '%';
      document.getElementById('pm25-val').textContent      = (q.pm2_5 || '--') + ' µg';
      document.getElementById('pm10-val').textContent      = (q.pm10  || '--') + ' µg';
      const cat = aqiCategory(aqi);
      const badge = document.getElementById('aqi-badge');
      badge.textContent = cat.label;
      badge.className   = 'badge ' + cat.cls;
      // Update AQI card location label
      const aqiLoc = document.getElementById('aqi-location');
      if (aqiLoc && label) aqiLoc.textContent = label + ' · Live Data';
    }

  } catch (err) {
    toast('❌ Could not fetch weather for this location', 'error');
    console.error('searchLocation error:', err);
  }
}

  function useMyLocation() {
  if (!navigator.geolocation) {
    toast('❌ Your browser does not support location detection', 'error');
    return;
  }
  const btn = document.getElementById("hero-use-location-btn");
  btn.innerHTML = '<span>📍</span> Locating...';
  navigator.geolocation.getCurrentPosition(
    function(pos) {
      const lat = pos.coords.latitude;
      const lon = pos.coords.longitude;
      btn.innerHTML = '<span>✅</span> Location found';
      fetch('https://api.bigdatacloud.net/data/reverse-geocode-client?latitude=' + lat + '&longitude=' + lon + '&localityLanguage=en')
        .then(r => r.json())
        .then(d => {
          const city = d.city || d.locality || d.principalSubdivision || "";
          const state = d.principalSubdivision || "";
          const label = city ? city + (state && state !== city ? ", " + state : "") : lat.toFixed(4) + ", " + lon.toFixed(4);
          document.getElementById('hero-location-search').value = label;
          const h = document.getElementById('location-search');
          if (h) h.value = label;
          toast('📍 Location detected: ' + label, 'success');
        })
        .catch(() => {
          document.getElementById('hero-location-search').value = lat.toFixed(4) + ", " + lon.toFixed(4);
        });
      if (typeof searchLocation === "function") {
        searchLocation(lat, lon);
      } else if (typeof loadWeather === "function") {
        loadWeather(lat, lon);
      }
      // update map card to GPS location
      fetch('https://api.bigdatacloud.net/data/reverse-geocode-client?latitude=' + lat + '&longitude=' + lon + '&localityLanguage=en')
        .then(function(r) { return r.json(); })
        .then(function(d) {
          var city = d.city || d.locality || d.principalSubdivision || '';
          var state = d.principalSubdivision || '';
          var lbl = city ? city + (state && state !== city ? ', ' + state : '') : lat.toFixed(4) + ', ' + lon.toFixed(4);
          if (typeof updateMapLocation === 'function') updateMapLocation(lat, lon, lbl);
        })
        .catch(function() {
          if (typeof updateMapLocation === 'function') updateMapLocation(lat, lon, lat.toFixed(4) + ', ' + lon.toFixed(4));
        });
    },
    function(err) {
      btn.innerHTML = '<span>🎯</span> Use My Location';
      if (err.code === 1) {
        // PERMISSION_DENIED
        toast('❌ Location blocked — allow in browser settings and retry', 'error');
      } else if (err.code === 2) {
        // POSITION_UNAVAILABLE
        toast('❌ Location unavailable — check GPS/network', 'error');
      } else {
        // TIMEOUT or other
        toast('❌ Location timed out — try again', 'error');
      }
    },
    { timeout: 10000, maximumAge: 60000 }
  );
}

  if (typeof loadWeather === "function") loadWeather();
};

// State
var _wsContacts = [];      // [{name, phone}]
var _wsUserName = '';
var _wsLat = null, _wsLon = null;
var _wsSOSActive = false;
var _wsWatchId = null;

// ── OPEN / CLOSE ──
function openWomenSafetyModal() {
  wsLoadState();
  wsRenderContacts();
  wsRefreshLocation();
  document.getElementById('ws-modal-overlay').style.display = 'flex';
  document.body.style.overflow = 'hidden';
}
function closeWomenSafetyModal() {
  // Only allow closing if no active SOS — or confirm
  if (_wsSOSActive) {
    if (!confirm('⚠️ An SOS is active! Are you sure you want to close? (Send "I\'m Safe" first if you\'re okay)')) return;
  }
  document.getElementById('ws-modal-overlay').style.display = 'none';
  document.body.style.overflow = '';
  // stop watching location
  if (_wsWatchId !== null) { navigator.geolocation.clearWatch(_wsWatchId); _wsWatchId = null; }
}
function wsHandleOverlayClick(e) {
  if (e.target === document.getElementById('ws-modal-overlay')) {
    closeWomenSafetyModal();
  }
}

// ── LOCATION ──
function wsRefreshLocation() {
  var lbl  = document.getElementById('ws-location-label');
  var coords = document.getElementById('ws-coords-label');
  lbl.textContent = '🔄 Locating…';

  if (!navigator.geolocation) {
    lbl.textContent = '⚠️ Geolocation not supported';
    return;
  }

  // Start continuous watch so location stays fresh
  if (_wsWatchId !== null) navigator.geolocation.clearWatch(_wsWatchId);
  _wsWatchId = navigator.geolocation.watchPosition(
    function(pos) {
      _wsLat = pos.coords.latitude;
      _wsLon = pos.coords.longitude;
      var acc = Math.round(pos.coords.accuracy);
      lbl.textContent = '📍 Location acquired';
      coords.textContent = _wsLat.toFixed(5) + ', ' + _wsLon.toFixed(5) + ' (±' + acc + 'm)';
    },
    function(err) {
      lbl.textContent = '⚠️ Could not get location — tap Refresh';
      coords.textContent = err.message;
    },
    { enableHighAccuracy: true, maximumAge: 10000, timeout: 15000 }
  );
}

function wsMapsLink() {
  if (_wsLat && _wsLon) {
    return 'https://www.google.com/maps?q=' + _wsLat + ',' + _wsLon;
  }
  return 'https://www.google.com/maps';
}

// ── CONTACTS ──
function wsAddContact() {
  var nameEl  = document.getElementById('ws-new-contact-name');
  var phoneEl = document.getElementById('ws-new-contact-phone');
  var name  = nameEl.value.trim();
  var phone = phoneEl.value.trim();

  if (!name || !phone) { wsToast('Enter both name and phone number', 'error'); return; }
  var digits = phone.replace(/^\+?91/, '').replace(/\D/g, '');
  if (digits.length !== 10) { wsToast('Enter a valid 10-digit Indian number', 'error'); return; }
  if (_wsContacts.length >= 5) { wsToast('Maximum 5 guardians allowed', 'error'); return; }

  // Check duplicate
  var fullPhone = '+91' + digits;
  for (var i = 0; i < _wsContacts.length; i++) {
    if (_wsContacts[i].phone === fullPhone) { wsToast('This number is already added', 'error'); return; }
  }

  _wsContacts.push({ name: name, phone: fullPhone });
  nameEl.value = ''; phoneEl.value = '';
  wsRenderContacts();
  wsSaveState();
  wsUpdateCardCounts();
  wsToast('✅ ' + name + ' added as guardian');
}

function wsRemoveContact(idx) {
  _wsContacts.splice(idx, 1);
  wsRenderContacts();
  wsSaveState();
  wsUpdateCardCounts();
}

function wsRenderContacts() {
  var list = document.getElementById('ws-contacts-list');
  list.innerHTML = '';

  if (_wsContacts.length === 0) {
    list.innerHTML = '<div style="text-align:center;color:var(--muted);font-size:.78rem;padding:12px 0;">No guardians added yet. Add trusted contacts above.</div>';
    return;
  }

  _wsContacts.forEach(function(c, i) {
    var chip = document.createElement('div');
    chip.className = 'ws-contact-chip';
    chip.innerHTML =
      '<div style="display:flex;align-items:center;gap:8px;">' +
        '<span style="font-size:1.1rem;filter:drop-shadow(0 0 4px rgba(244,114,182,.5))">👤</span>' +
        '<div>' +
          '<div style="font-size:.85rem;font-weight:600;color:var(--text);">' + wsEsc(c.name) + '</div>' +
          '<div style="font-size:.72rem;color:var(--muted);font-family:monospace;">' + c.phone + '</div>' +
        '</div>' +
      '</div>' +
      '<button onclick="wsRemoveContact(' + i + ')" ' +
              'style="background:rgba(255,71,87,.12);border:1px solid rgba(255,71,87,.3);' +
              'color:var(--red);border-radius:8px;padding:4px 10px;cursor:pointer;' +
              'font-size:.72rem;font-weight:700;">Remove</button>';
    list.appendChild(chip);
  });

  // Hide add row if at max
  document.getElementById('ws-add-row').style.display = _wsContacts.length >= 5 ? 'none' : 'flex';
}

// ── SOS TRIGGER ──
function wsTriggerSOS() {
  var userName = document.getElementById('ws-user-name').value.trim() || _wsUserName || 'AlertIndia User';

  if (_wsContacts.length === 0) {
    wsToast('⚠️ Add at least one guardian contact first', 'error');
    return;
  }
  if (!_wsLat || !_wsLon) {
    wsToast('⚠️ Location not yet available — tap Refresh and wait', 'error');
    return;
  }

  if (!confirm('🚨 SEND SOS ALERT?\n\nThis will send your live location to ' + _wsContacts.length + ' guardian(s) and the local police station.\n\nConfirm only if you genuinely need help.')) return;

  _wsSOSActive = true;
  _wsUserName  = userName;

  var mapsLink = wsMapsLink();
  var locStr   = _wsLat.toFixed(5) + ', ' + _wsLon.toFixed(5);

  // Message to personal contacts
  var personalMsg =
    '🚨 EMERGENCY SOS from AlertIndia 🚨\n\n' +
    '👤 *' + userName + '* may be in DANGER and needs immediate help!\n\n' +
    '📍 *Live Location:* ' + mapsLink + '\n' +
    '🗺️ Coordinates: ' + locStr + '\n\n' +
    '⚡ Please call them or send help NOW.\n' +
    'If unreachable — call police: 100\n\n' +
    '— AlertIndia Guardian Shield';

  // Message to police / cyber
  var policeMsg =
    '🚨 EMERGENCY ALERT — AlertIndia SOS System 🚨\n\n' +
    'Person in distress: *' + userName + '*\n' +
    '📍 Location: ' + mapsLink + '\n' +
    'Coordinates: ' + locStr + '\n\n' +
    'This person has activated an emergency SOS via AlertIndia.\n' +
    'Please send police assistance immediately.\n\n' +
    '— AlertIndia Earliest Warning System';

  // Send to all guardian contacts via backend
  var sendPromises = [];
  _wsContacts.forEach(function(c) {
    sendPromises.push(
      fetch(GAS_URL, {
        method: 'POST', mode: 'no-cors',
        body: JSON.stringify({
          action: 'ws_sos',
          to_phone: c.phone,
          to_name: c.name,
          user_name: userName,
          message: personalMsg,
          lat: _wsLat,
          lon: _wsLon,
          maps_link: mapsLink
        })
      }).catch(function(e){ console.warn('SOS send error:', e); })
    );
  });

  // Also notify police helpline (100 as placeholder — GAS routes this)
  sendPromises.push(
    fetch(GAS_URL, {
      method: 'POST', mode: 'no-cors',
      body: JSON.stringify({
        action: 'ws_police_notify',
        user_name: userName,
        message: policeMsg,
        lat: _wsLat,
        lon: _wsLon,
        maps_link: mapsLink
      })
    }).catch(function(e){ console.warn('Police notify error:', e); })
  );

  Promise.allSettled(sendPromises).then(function() {
    console.log('SOS dispatched to ' + sendPromises.length + ' recipients');
  });

  // Update UI
  var sosBtn = document.getElementById('ws-sos-btn');
  sosBtn.style.background = 'linear-gradient(135deg,#7f1d1d,#dc2626)';
  sosBtn.textContent = '🔴 SOS ACTIVE — Help is coming';
  sosBtn.classList.add('ws-sos-armed');

  // Show ring
  var ring = document.getElementById('ws-sos-ring');
  ring.style.display = 'block';
  setTimeout(function(){ ring.style.display = 'none'; }, 1100);

  var statusEl = document.getElementById('ws-sos-status');
  statusEl.style.display = 'block';
  statusEl.innerHTML =
    '✅ SOS sent to <b>' + _wsContacts.length + ' guardian(s)</b> + local police station.<br>' +
    '📍 Your location: <a href="' + mapsLink + '" target="_blank" style="color:#f472b6;">' + locStr + '</a>';

  document.getElementById('ws-safe-section').style.display = 'block';

  // Update card
  document.getElementById('ws-status-card').textContent = '🔴 SOS ACTIVE';
  document.getElementById('ws-status-card').style.color = '#f43f5e';
  document.getElementById('ws-card-badge').textContent  = 'SOS ACTIVE';
  document.getElementById('ws-card-badge').style.color  = '#f43f5e';
  document.getElementById('ws-card-pulse').style.display = 'block';

  wsToast('🚨 SOS sent to all guardians + police', 'error');
  wsSaveState();
}

// ── SAFE / CANCEL ──
function wsSendSafe() {
  if (!confirm('✅ Confirm you are safe?\n\nThis will send a "all clear" message to all your guardian contacts.')) return;

  var userName = _wsUserName || document.getElementById('ws-user-name').value.trim() || 'AlertIndia User';
  var safeMsg  =
    '✅ ALL CLEAR from AlertIndia\n\n' +
    '👤 *' + userName + '* is SAFE and okay!\n\n' +
    'The earlier SOS alert was either a false alarm or the situation has been resolved.\n' +
    'No further action needed — thank you for your concern! 🙏\n\n' +
    '— AlertIndia Guardian Shield';

  _wsContacts.forEach(function(c) {
    fetch(GAS_URL, {
      method: 'POST', mode: 'no-cors',
      body: JSON.stringify({
        action: 'ws_safe',
        to_phone: c.phone,
        to_name: c.name,
        user_name: userName,
        message: safeMsg
      })
    }).catch(function(e){ console.warn('Safe send error:', e); });
  });

  _wsSOSActive = false;

  // Reset UI
  var sosBtn = document.getElementById('ws-sos-btn');
  sosBtn.style.background = 'linear-gradient(135deg,#be185d,#f43f5e)';
  sosBtn.textContent      = '🚨  I NEED HELP — Send SOS Now';
  sosBtn.classList.remove('ws-sos-armed');

  document.getElementById('ws-sos-status').style.display  = 'none';
  document.getElementById('ws-safe-section').style.display = 'none';

  document.getElementById('ws-status-card').textContent = 'Inactive';
  document.getElementById('ws-status-card').style.color = 'var(--muted)';
  document.getElementById('ws-card-badge').textContent  = 'STANDBY';
  document.getElementById('ws-card-badge').style.color  = '#f472b6';
  document.getElementById('ws-card-pulse').style.display = 'none';

  wsToast('✅ Safe message sent to all guardians', 'success');
  wsSaveState();
}

// ── PERSIST STATE (sessionStorage — clears on tab close) ──
function wsSaveState() {
  _wsUserName = document.getElementById('ws-user-name').value.trim();
  try {
    sessionStorage.setItem('ws_state', JSON.stringify({
      contacts: _wsContacts,
      userName: _wsUserName,
      sosActive: _wsSOSActive
    }));
  } catch(e){}
}
function wsLoadState() {
  try {
    var raw = sessionStorage.getItem('ws_state');
    if (!raw) return;
    var s = JSON.parse(raw);
    _wsContacts = s.contacts  || [];
    _wsUserName  = s.userName || '';
    _wsSOSActive = s.sosActive || false;
    document.getElementById('ws-user-name').value = _wsUserName;
    // Restore SOS UI if it was active
    if (_wsSOSActive) {
      var sosBtn = document.getElementById('ws-sos-btn');
      sosBtn.style.background = 'linear-gradient(135deg,#7f1d1d,#dc2626)';
      sosBtn.textContent = '🔴 SOS ACTIVE — Help is coming';
      sosBtn.classList.add('ws-sos-armed');
      document.getElementById('ws-sos-status').style.display = 'block';
      document.getElementById('ws-sos-status').innerHTML = '⚡ SOS is still active. Send "I\'m Safe" when resolved.';
      document.getElementById('ws-safe-section').style.display = 'block';
    }
    wsUpdateCardCounts();
  } catch(e){}
}

function wsUpdateCardCounts() {
  var el = document.getElementById('ws-contact-count-card');
  if (el) el.textContent = _wsContacts.length + ' saved';
}

// ── HELPERS ──
function wsEsc(str) {
  return String(str).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}
function wsToast(msg, type) {
  if (typeof toast === 'function') { toast(msg, type || 'success'); return; }
  var t = document.getElementById('toast');
  if (t) { t.textContent = msg; t.className = 'toast ' + (type||'success') + ' show'; setTimeout(function(){ t.className='toast'; }, 3200); }
}

window.addEventListener('load', function() {
  setTimeout(function() {
    try {
      var raw = sessionStorage.getItem('ws_state');
      if (raw) { var s = JSON.parse(raw); _wsContacts = s.contacts||[]; wsUpdateCardCounts(); }
    } catch(e){}
  }, 500);
});
